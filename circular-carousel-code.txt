Replace the <script> section (lines 1589-1661) with this:

                    <script>
                        (function () {
                            function initSlider(root) {
                                var row = root.querySelector('.row');
                                if (!row) return;
                                var items = Array.prototype.slice.call(row.children).filter(function (n) { return n && n.nodeType === 1; });
                                var cards = items.map(function (el) { return el.querySelector('.card'); }).filter(Boolean);
                                var activeIndex = 0;
                                var totalItems = items.length;

                                function positionCards() {
                                    // 3D Coverflow effect - like the image
                                    var spacing = 280; // Space between cards
                                    var maxVisible = 2; // Max cards visible on each side
                                    
                                    items.forEach(function (item, i) {
                                        var relativeIndex = (i - activeIndex + totalItems) % totalItems;
                                        
                                        // Normalize to -totalItems/2 to +totalItems/2
                                        if (relativeIndex > totalItems / 2) {
                                            relativeIndex = relativeIndex - totalItems;
                                        }
                                        
                                        // Active card in center
                                        if (relativeIndex === 0) {
                                            item.style.transform = 'translate(-50%, -50%) translateX(0px) translateY(0px) translateZ(0px) rotateY(0deg) scale(1)';
                                            item.style.zIndex = 50;
                                            item.style.opacity = 1;
                                        } else {
                                            // Calculate position for side cards
                                            var x = relativeIndex * spacing;
                                            var z = -Math.abs(relativeIndex) * 150; // Push back based on distance
                                            var rotateY = relativeIndex > 0 ? -35 : 35; // Rotate cards
                                            var scale = 1 - Math.abs(relativeIndex) * 0.15; // Scale down
                                            var opacity = Math.max(0.3, 1 - Math.abs(relativeIndex) * 0.2);
                                            
                                            // Hide cards beyond maxVisible
                                            if (Math.abs(relativeIndex) > maxVisible) {
                                                opacity = 0;
                                                item.style.pointerEvents = 'none';
                                            } else {
                                                item.style.pointerEvents = 'auto';
                                            }
                                            
                                            item.style.transform = 'translate(-50%, -50%) translateX(' + x + 'px) translateY(0px) translateZ(' + z + 'px) rotateY(' + rotateY + 'deg) scale(' + scale + ')';
                                            item.style.zIndex = Math.round(50 - Math.abs(relativeIndex) * 10);
                                            item.style.opacity = opacity;
                                        }
                                    });
                                }

                                function setActive(index) {
                                    // Allow wrapping
                                    if (index < 0) index = totalItems - 1;
                                    if (index >= totalItems) index = 0;
                                    activeIndex = index;
                                    
                                    cards.forEach(function (c) { c.classList.remove('is-active'); });
                                    if (cards[activeIndex]) cards[activeIndex].classList.add('is-active');
                                    
                                    positionCards();
                                }

                                // Initialize
                                setActive(0);

                                var prev = root.querySelector('.slider-prev');
                                var next = root.querySelector('.slider-next');
                                
                                if (prev) prev.addEventListener('click', function () {
                                    setActive(activeIndex - 1);
                                });
                                
                                if (next) next.addEventListener('click', function () {
                                    setActive(activeIndex + 1);
                                });

                                // Click to focus
                                row.addEventListener('click', function (e) {
                                    var card = e.target && e.target.closest('.card');
                                    if (!card) return;
                                    var idx = cards.indexOf(card);
                                    if (idx === -1) return;
                                    setActive(idx);
                                });
                                
                                // Keyboard navigation
                                document.addEventListener('keydown', function(e) {
                                    if (e.key === 'ArrowLeft') setActive(activeIndex - 1);
                                    if (e.key === 'ArrowRight') setActive(activeIndex + 1);
                                });
                                
                                window.addEventListener('resize', positionCards);
                            }
                            document.addEventListener('DOMContentLoaded', function () {
                                document.querySelectorAll('.tickets-slider').forEach(initSlider);
                            });
                        })();
                    </script>
